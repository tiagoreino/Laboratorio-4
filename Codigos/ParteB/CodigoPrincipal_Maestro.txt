/* Maestro - ATmega328P.
 * AVR-GCC / Microchip Studio / 16 MHz.
 */
#define F_CPU 16000000UL
#include <avr/io.h>
#include <avr/interrupt.h>
#include <avr/wdt.h>
#include <util/atomic.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#define LCD_ADDR_7BIT       0x27
#define LCD_BACKLIGHT_HIGH     1
#define NTC_TO_GND             1
#define NTC_TRIP_ADC         204U
#define NTC_SAFE_ADC         235U
#define LDR_DETECT_LOW         1
#define LDR_LOW              400U
#define LDR_HIGH             500U
#define SENSOR_PERIOD_MS      10UL
#define DEBOUNCE_MS           30UL
#define SPI_MAX_ATTEMPTS       3
#define SPI_REPLY_WAIT_MS      2UL
#define SPI_BYTE_TIMEOUT_MS    5UL
#define SPI_HEARTBEAT_MS      80UL
#define JAM_TIMEOUT_MS      4000UL

enum { C_SYNC=1,C_START,C_STOP,C_EMERG,C_PWM,C_PISTON,C_STATUS };
enum { S_STOP,S_RUN,S_PISTON,S_THERMAL,S_LINK,S_EMERGENCY,S_JAM };
static volatile uint32_t ticks;
static volatile uint8_t tx_head,tx_tail;
static char tx_ring[256];
static uint8_t state=S_STOP, thermal, sensor_fault, ready_sensors;
static uint8_t estop_latched;
static uint8_t thermal_latched, jam_latched, jam_watching;
static uint32_t jam_since;
static uint8_t link_ok, need_sync=1, want_start, urgent, sequence;
static uint8_t pwm_target, pwm_sent, pwm_valid, piece, piece_armed=1, piece_pending;
static uint8_t btn_raw,btn_stable,ldr_raw,ldr_stable;
static uint16_t analog[3];
static uint32_t count, sensor_at,btn_at,ldr_at,heartbeat_at,recover_at;
static const char *last_command="BOOT";
static uint8_t debug_cmd,debug_rx0,debug_rx1;

/* TIMER2_COMPA_vect — Interrupción de comparación del Timer2: incrementa ticks cada 1 ms.
 * Sin parámetros ni retorno. Proporciona la base temporal usada por las tareas; la lectura de 32 bits se protege en now_ms.
 */
ISR(TIMER2_COMPA_vect) { ++ticks; }
/* USART_UDRE_vect — Interrupción de registro de transmisión UART vacío: envía el siguiente carácter de tx_ring.
 * Sin parámetros ni retorno. Avanza tx_tail; si la cola está vacía, deshabilita esta interrupción hasta que uart_text agregue datos.
 */
ISR(USART_UDRE_vect)
{
    if (tx_tail==tx_head) UCSR0B&=(uint8_t)~_BV(UDRIE0);
    else { UDR0=tx_ring[tx_tail]; ++tx_tail; }
}
/* now_ms — Obtiene una copia coherente del contador de milisegundos.
 * Entrada: ninguna. Retorno: ticks como uint32_t. El bloque atómico evita que la ISR modifique parte del contador durante su lectura y restaura el estado previo de interrupciones.
 */
static uint32_t now_ms(void)
{
    uint32_t n; ATOMIC_BLOCK(ATOMIC_RESTORESTATE) { n=ticks; } return n;
}
/* due — Comprueba si se alcanzó un instante programado.
 * Entradas: n, tiempo actual; d, vencimiento, ambos en ms. Retorna 1 si n-d interpretado con signo es no negativo; 0 en otro caso. La comparación permite cruzar el desbordamiento para plazos separados por menos de 2^31 ms.
 */
static uint8_t due(uint32_t n,uint32_t d) { return (int32_t)(n-d)>=0; }
/* uart_text — Encola una cadena para transmitirla mediante la ISR UART, sin esperar al puerto.
 * Entrada: s, cadena terminada en cero. Sin retorno. Si no cabe completa en la cola circular de 256 posiciones, descarta el informe entero. Publica el nuevo índice y habilita la interrupción dentro de un bloque atómico.
 */
static void uart_text(const char *s)
{
    uint8_t h=tx_head,free_bytes=(uint8_t)(tx_tail-h-1);
    uint16_t len=strlen(s);
    if (len>free_bytes) return; /* Descartar informe entero, nunca bloquear. */
    while (*s) { tx_ring[h]=*s++; ++h; }
    ATOMIC_BLOCK(ATOMIC_RESTORESTATE) { tx_head=h; UCSR0B|=_BV(UDRIE0); }
}

/* ADC asincrono, tres canales con descarte tras cada cambio de multiplexor. */
/* sensors — Avanza la adquisición ADC de LDR (canal 0), NTC (1) y potenciómetro (2).
 * Entrada: n, tiempo actual en ms. Sin retorno. Inicia conversiones y consulta su finalización sin espera activa; descarta la primera conversión tras cambiar canal. Inicializa cada lectura y después aplica un filtro de 3/4 del valor previo y 1/4 del nuevo. Detecta fallo NTC con el valor crudo; al completar los tres canales habilita ready_sensors y programa la siguiente ronda.
 */
static void sensors(uint32_t n)
{
    static uint8_t channel,phase,discard=1,initialized[3];
    /* ADC: solo inicia una ronda al vencer sensor_at; las llamadas siguientes consultan ADSC. */
    if (!phase) {
        if (!due(n,sensor_at)) return;
        ADMUX=_BV(REFS0)|channel; ADCSRA|=_BV(ADSC); phase=1; return;
    }
    if (ADCSRA&_BV(ADSC)) return;
    { uint16_t v=ADC;
      /* Tras seleccionar canal se descarta una conversión para permitir el asentamiento del multiplexor. */
      if (discard) { discard=0; ADCSRA|=_BV(ADSC); return; }
      if (!initialized[channel]) { analog[channel]=v; initialized[channel]=1; }
      else analog[channel]=(uint16_t)((3UL*analog[channel]+v+2)/4);
      /* Averia NTC por lectura cruda: no esperar al filtro. */
      if (channel==1) sensor_fault=(v<=3 || v>=1020);
    }
    discard=1; phase=0;
    if (++channel==3) { channel=0; ready_sensors=1; sensor_at=n+SENSOR_PERIOD_MS; }
}
/* ntc_adc — Normaliza la lectura filtrada de la NTC según la orientación de su divisor.
 * Entrada: ninguna; utiliza analog[1] y NTC_TO_GND. Retorno: ADC normalizado de 0 a 1023; con la configuración actual devuelve analog[1], y con la orientación inversa devuelve 1023-analog[1].
 */
static uint16_t ntc_adc(void) { return NTC_TO_GND ? analog[1] : 1023-analog[1]; }
/* temperature — Estima la temperatura de la NTC en grados Celsius enteros.
 * Entrada: ninguna; consulta ntc_adc. Retorno: int16_t calculado mediante tabla e interpolación lineal entre puntos, limitado a los extremos -20 y 100. No usa punto flotante; las alarmas se deciden por umbrales ADC en inputs.
 */
static int16_t temperature(void)
{
    /* Tabla ADC descendente y temperaturas correspondientes; la interpolación usa aritmética entera. */
    static const uint16_t a[]={934,788,684,569,512,455,354,270,235,204,153,115,67};
    static const int16_t t[]={-20,0,10,20,25,30,40,50,55,60,70,80,100};
    uint16_t v=ntc_adc(); uint8_t i;
    if(v>=a[0]) return t[0];
    for(i=1;i<13;i++) if(v>=a[i])
        return t[i-1]+(int32_t)(a[i-1]-v)*(t[i]-t[i-1])/(a[i-1]-a[i]);
    return 100;
}
/* emergency_pressed — Lee el pulsador de emergencia conectado a PD3, activo a nivel bajo.
 * Entrada: ninguna. Retorno: 1 con el pin a GND y 0 con nivel alto. Esta lectura directa no aplica antirrebote.
 */
static uint8_t emergency_pressed(void) { return !(PIND&_BV(PD3)); }
/* stopped_reason — Selecciona el estado de parada que corresponde a las condiciones actuales.
 * Entrada: ninguna; consulta alarmas enclavadas, emergencia física, condición térmica y enlace SPI. Retorno: S_EMERGENCY, S_THERMAL, S_JAM, S_LINK o S_STOP, en ese orden de prioridad. No borra alarmas.
 */
static uint8_t stopped_reason(void)
{
    /* Las alarmas conservan su prioridad hasta el rearme manual. */
    if(estop_latched || emergency_pressed()) return S_EMERGENCY;
    if(thermal_latched || thermal) return S_THERMAL;
    if(jam_latched) return S_JAM;
    if(!link_ok || need_sync) return S_LINK;
    return S_STOP;
}
/* safety_latched — Indica si hay alguna alarma de seguridad enclavada.
 * Entrada: ninguna. Retorno: 1 si emergencia, temperatura o atasco están enclavados; 0 si ninguno lo está. El fallo de enlace se evalúa por separado.
 */
static uint8_t safety_latched(void)
{
    return estop_latched || thermal_latched || jam_latched;
}
/* halt — Aplica una parada local y deja pendiente la orden de parada para el esclavo.
 * Entrada: reason, estado que debe mostrarse. Sin retorno. Cancela solicitud START, pieza pendiente y vigilancia de atasco. Selecciona C_EMERG si existe una alarma de seguridad o el motivo es térmico, emergencia o atasco; en otro caso selecciona C_STOP.
 */
static void halt(uint8_t reason)
{
    state=reason; want_start=0; piece_pending=0; jam_watching=0;
    urgent=(safety_latched() || reason==S_THERMAL ||
            reason==S_EMERGENCY || reason==S_JAM ? C_EMERG:C_STOP);
}
/* inputs — Actualiza entradas, velocidad solicitada, detección de piezas y alarmas; evalúa el rearme manual.
 * Entrada: n, tiempo actual en ms. Sin retorno. Atiende emergencia primero; procesa NTC con histéresis, LDR con histéresis y antirrebote de 30 ms, y atasco tras la confirmación del retorno. START requiere un nuevo flanco estable y todas las condiciones de rearme; liberar un sensor por sí solo no borra el enclavamiento.
 */
static void inputs(uint32_t n)
{
    uint8_t b=!(PIND&_BV(PD2)),d;
    if(emergency_pressed() && !estop_latched) {
        estop_latched=1; halt(stopped_reason());
    }
    if(ready_sensors) {
        /* NTC: fallo crudo o umbral de disparo activa alarma; el umbral de recuperación solo limpia thermal, no el enclavamiento. */
        if(sensor_fault || ntc_adc()<=NTC_TRIP_ADC) {
            if(!thermal) {
                thermal=1; thermal_latched=1; halt(stopped_reason());
            }
        } else if(ntc_adc()>=NTC_SAFE_ADC) thermal=0;
        /* Escala el potenciómetro filtrado de 0..1023 a PWM de 0..255, con redondeo. */
        pwm_target=(uint8_t)((analog[2]*255UL+511)/1023);
        /* LDR: los dos umbrales forman una histéresis; dentro de la banda se conserva el estado previo. */
        d=ldr_raw;
        if(LDR_DETECT_LOW) {
            if(analog[0]<=LDR_LOW) d=1; else if(analog[0]>=LDR_HIGH) d=0;
        } else {
            if(analog[0]>=LDR_HIGH) d=1; else if(analog[0]<=LDR_LOW) d=0;
        }
        /* Antirrebote LDR: exige 30 ms sin cambios antes de actualizar piece. Otra pieza requiere una liberación estable previa. */
        if(d!=ldr_raw) { ldr_raw=d; ldr_at=n; }
        if(d!=ldr_stable && (uint32_t)(n-ldr_at)>=DEBOUNCE_MS) {
            ldr_stable=d; piece=d;
            if(!d) piece_armed=1;
            else if(piece_armed) {
                piece_armed=0;
                if(state==S_RUN) piece_pending=1;
            }
        }
        /* Solo se habilita al confirmar por STATUS que el piston termino.
         * Una liberacion cancela, no pausa: no sumar intervalos separados. */
        /* Atasco: cuenta 4000 ms consecutivos de pieza presente, únicamente mientras sigue S_RUN. Liberar la LDR cancela esta vigilancia. */
        if(jam_watching) {
            if(state!=S_RUN || !piece) jam_watching=0;
            else if((uint32_t)(n-jam_since)>=JAM_TIMEOUT_MS) {
                jam_latched=1; halt(stopped_reason());
            }
        }
    }
    if(safety_latched()) state=stopped_reason();
    /* Evaluar START despues de actualizar TODAS las condiciones de alarma.
     * Solo un nuevo flanco, nunca el boton mantenido, puede rearmar. */
    /* Antirrebote START: mide estabilidad y actúa al pasar a pulsado; mantenerlo presionado no genera otro rearme. */
    if(b!=btn_raw) { btn_raw=b; btn_at=n; }
    if(b!=btn_stable && (uint32_t)(n-btn_at)>=DEBOUNCE_MS) {
        btn_stable=b;
        if(b && ready_sensors && state!=S_RUN && state!=S_PISTON &&
           !want_start && !emergency_pressed() && !thermal &&
           (!jam_latched || !piece) && link_ok && !need_sync) {
            /* Solo este START aceptado borra las alarmas enclavadas y solicita un nuevo arranque con PWM previamente confirmado. */
            estop_latched=0; thermal_latched=0; jam_latched=0;
            jam_watching=0; state=S_STOP;
            want_start=1; pwm_valid=0;
        }
    }
}

/* SPI: comando y consulta separados por SS. Cada paso retorna al main. */
static struct {
    uint8_t phase,cmd,val,attempt,result,rx0,rx1,query;
    uint32_t at;
} bus;
/* spi_restore — Libera SS y reinicializa el periférico SPI maestro.
 * Entrada: ninguna. Sin retorno. Deshabilita SPI, lee SPSR y SPDR para limpiar su estado y configura modo 1, reloj F_CPU/128 (125 kHz a 16 MHz), sin doble velocidad.
 */
static void spi_restore(void)
{
    PORTB|=_BV(PB2);
    SPCR=0;
    { volatile uint8_t d=SPSR; d=SPDR; (void)d; }
    SPCR=_BV(SPE)|_BV(MSTR)|_BV(CPHA)|_BV(SPR1)|_BV(SPR0); SPSR=0;
}
/* spi_send — Prepara un intercambio SPI de comando y valor, ejecutado después por spi_task.
 * Entradas: c, comando; v, valor; n, tiempo actual en ms. Sin retorno. Alterna el bit de secuencia 0x10, inicializa intento y resultado, activa SS y programa el comienzo del primer byte a n+1.
 */
static void spi_send(uint8_t c,uint8_t v,uint32_t n)
{
    sequence^=0x10; bus.cmd=c|sequence; bus.val=v;
    bus.attempt=1; bus.result=0; bus.query=0;
    PORTB&=(uint8_t)~_BV(PB2); bus.at=n+1; bus.phase=1;
}
/* spi_fail — Gestiona un timeout o una respuesta inválida de la transacción SPI.
 * Entrada: n, tiempo actual en ms. Sin retorno. Restaura SPI y programa un reintento tras 4 ms mientras queden intentos; al agotarlos deja el bus inactivo y bus.result=2.
 */
 static void spi_fail(uint32_t n)
{
    spi_restore();
    if(bus.attempt++<SPI_MAX_ATTEMPTS) { bus.phase=5; bus.at=n+4; }
    else { bus.phase=0; bus.result=2; }
}
/* spi_task — Avanza la máquina de fases que envía dos bytes y consulta después su confirmación SPI.
 * Entrada: n, tiempo actual en ms. Sin retorno. Cada llamada procesa un paso y vuelve al main. Separa comando y consulta mediante SS, respeta pausas y timeout por byte, y verifica el ACK con el comando y su secuencia. Para STATUS exige que el nibble alto de estado sea cero; para otros comandos compara el valor devuelto. Resultado: bus.result=1 al confirmar o 2 al agotar reintentos.
 */
static void spi_task(uint32_t n)
{
    uint8_t r;
    /* Fases SPI: 0 reposo; 1 primer byte; 2 espera; 7 pausa entre bytes; 3 espera segundo byte; 6 cierre; 4 espera ACK; 5 reintento. */
    switch(bus.phase) {
    case 0: break;
    case 1:
        if(!due(n,bus.at)) break;
        SPDR=bus.query ? 0:bus.cmd; bus.at=n+SPI_BYTE_TIMEOUT_MS; bus.phase=2; break;
    case 2:
        if(!(SPSR&_BV(SPIF))) { if(due(n,bus.at)) spi_fail(n); break; }
        r=SPDR; if(bus.query) bus.rx0=r;
        /* Dar tiempo a SPI_STC del esclavo para precargar su segundo byte. */
        bus.at=n+1; bus.phase=7; break;
    case 7:
        if(!due(n,bus.at)) break;
        SPDR=bus.query ? 0:bus.val; bus.at=n+SPI_BYTE_TIMEOUT_MS; bus.phase=3; break;
    case 3:
        if(!(SPSR&_BV(SPIF))) { if(due(n,bus.at)) spi_fail(n); break; }
        r=SPDR; if(bus.query) bus.rx1=r;
        /* Conservar SS 1 ms mas permite atender SPI_STC en el esclavo
         * antes de la interrupcion de fin de seleccion. */
        bus.phase=6; bus.at=n+1; break;
    case 6:
        if(!due(n,bus.at)) break;
        PORTB|=_BV(PB2);
        if(!bus.query) { bus.phase=4; bus.at=n+SPI_REPLY_WAIT_MS; }
        else {
            debug_cmd=bus.cmd; debug_rx0=bus.rx0; debug_rx1=bus.rx1;
            /* Valida el ACK del intercambio actual, incluido el bit de secuencia; una respuesta inválida provoca reintento. */
            if(bus.rx0==(uint8_t)(0x80|bus.cmd) &&
               (((bus.cmd&15)==C_STATUS) ? !(bus.rx1&0xF0) : bus.rx1==bus.val)) {
                bus.phase=0; bus.result=1;
            } else spi_fail(n);
        } break;
    case 4:
        if(due(n,bus.at)) {
            bus.query=1; PORTB&=(uint8_t)~_BV(PB2); bus.phase=1; bus.at=n+1;
        } break;
    case 5:
        if(due(n,bus.at)) {
            bus.query=0; PORTB&=(uint8_t)~_BV(PB2); bus.phase=1; bus.at=n+1;
        } break;
    }
}
/* command_name — Traduce un código de comando a su nombre para los informes.
 * Entrada: c, código base del comando. Retorno: puntero a una cadena constante; devuelve ESTADO para cualquier código no enumerado en el switch.
 */
static const char *command_name(uint8_t c)
{
    switch(c) {
    case C_SYNC:return "SYNC"; case C_START:return "START";
    case C_STOP:return "STOP"; case C_EMERG:return "EMERG";
    case C_PWM:return "PWM"; case C_PISTON:return "PISTON";
    default:return "ESTADO";
    }
}
/* control — Coordina los estados del proceso y decide la próxima orden hacia el esclavo.
 * Entrada: n, tiempo actual en ms. Sin retorno. Consume el resultado SPI, confirma arranque y pistón, cuenta al finalizar el ciclo y habilita la vigilancia de atasco. Prioriza órdenes urgentes, sincronización, arranque, pieza pendiente, consulta periódica y actualización PWM; el fallo de enlace exige sincronización y una nueva solicitud START.
 */
static void control(uint32_t n)
{
    if(bus.result) {
        uint8_t c=bus.cmd&15;
        /* Fallo SPI definitivo: cancela arranque/pieza, conserva prioridad de alarmas y programa recuperación por SYNC. */
        if(bus.result==2) {
            link_ok=0; need_sync=1; want_start=0; piece_pending=0;
            jam_watching=0; state=stopped_reason(); recover_at=n+200;
            last_command="ERROR SPI";
        } else {
            link_ok=1;
            /* Consultar estado con cadencia propia aunque el potenciometro
             * cambie continuamente; asi siempre se observa fin de piston. */
            if(c==C_STATUS || c==C_SYNC) heartbeat_at=n+SPI_HEARTBEAT_MS;
            if(c!=C_STATUS) last_command=command_name(c);
            if(c==C_SYNC) {
                need_sync=0; pwm_valid=0;
                if(thermal || safety_latched()) urgent=C_EMERG;
            }
            if(c==C_PWM) { pwm_sent=bus.val; pwm_valid=1; }
            if(c==C_START) {
                want_start=0;
                if(!urgent && !thermal && !safety_latched() &&
                   !emergency_pressed()) state=S_RUN;
            }
            if(c==C_PISTON && !urgent && !thermal && !safety_latched()) {
                state=S_PISTON; jam_watching=0;
            }
            /* STATUS: bits 2/3 indican anomalía; bit 1 indica pistón ocupado y bit 0 marcha. Se cuenta al confirmar retorno y marcha. */
            if(c==C_STATUS) {
                if((bus.rx1&12) && (state==S_RUN || state==S_PISTON)) {
                    link_ok=0; need_sync=1; halt(stopped_reason()); recover_at=n+100;
                } else if(state==S_PISTON && !(bus.rx1&2) && (bus.rx1&1)) {
                    ++count; state=S_RUN;
                    /* La banda ya se reanudo localmente en el esclavo.
                     * El plazo empieza al recibir esta confirmacion. */
                    jam_watching=piece; jam_since=n;
                } else if(state==S_RUN && !(bus.rx1&1)) {
                    need_sync=1; halt(stopped_reason()); recover_at=n+100;
                }
            }
        }
        bus.result=0;
    }
    /* Una orden de seguridad puede cancelar un intercambio previo. */
    if(urgent && bus.phase && (bus.cmd&15)!=urgent) {
        spi_restore(); bus.phase=0; bus.result=0;
    }
    /* Solo se inicia otra orden cuando el intercambio está libre; la siguiente selección conserva la prioridad original. */
    if(bus.phase) return;
    if(urgent) { uint8_t c=urgent; urgent=0; spi_send(c,0,n); return; }
    if(need_sync) {
        if(due(n,recover_at)) spi_send(C_SYNC,0,n);
        return;
    }
    if(!ready_sensors) return;
    if(want_start && !thermal && !safety_latched() && !emergency_pressed()) {
        /* Enviar velocidad antes de START, evitando arranque con PWM viejo. */
        if(!pwm_valid) spi_send(C_PWM,pwm_target,n);
        else spi_send(C_START,0,n);
    } else if(piece_pending && state==S_RUN) {
        piece_pending=0; spi_send(C_PISTON,0,n);
    } else if(due(n,heartbeat_at)) spi_send(C_STATUS,0,n);
    else if(!pwm_valid || pwm_sent!=pwm_target) spi_send(C_PWM,pwm_target,n);
}

/* TWI de un byte al PCF: timeout, STOP y recuperacion con nueve pulsos SCL.
 * La recuperacion es temporizada por main y no frena control ni UART/SPI.
 */
static struct { uint8_t phase,value,ok,pulses; uint32_t at; } twi;
/* twi_start — Inicia el envío I2C de un byte al expansor PCF8574 de la LCD.
 * Entradas: v, byte de salida del expansor; n, tiempo actual en ms. Sin retorno. Guarda el byte, borra el indicador de éxito, solicita START y fija un timeout inicial de 4 ms; twi_task completa el envío.
 */
static void twi_start(uint8_t v,uint32_t n)
{
    twi.value=v; twi.ok=0; twi.phase=1; twi.at=n+4;
    TWCR=_BV(TWINT)|_BV(TWSTA)|_BV(TWEN);
}
/* twi_abort — Abandona una transferencia TWI e inicia la recuperación del bus I2C.
 * Entrada: n, tiempo actual en ms. Sin retorno. Deshabilita TWI, deja SDA y SCL como entradas sin pull-up interno y programa la primera fase de recuperación para n+1. Los niveles altos dependen de los pull-ups externos.
 */
static void twi_abort(uint32_t n)
{
    TWCR=0; PORTC&=(uint8_t)~(_BV(PC4)|_BV(PC5));
    DDRC&=(uint8_t)~(_BV(PC4)|_BV(PC5));
    twi.pulses=0; twi.phase=5; twi.at=n+1;
}
/* twi_task — Avanza el envío TWI o la recuperación temporizada del bus I2C.
 * Entrada: n, tiempo actual en ms. Sin retorno. Comprueba códigos de START, ACK de dirección y ACK de dato; solicita STOP y espera su finalización. Ante error o timeout llama a twi_abort. En recuperación alterna SCL entre salida baja y entrada durante nueve pulsos, libera SDA/SCL y reactiva TWI con twi.ok=0; solo una transferencia completada deja twi.ok=1.
 */
static void twi_task(uint32_t n)
{
    uint8_t s=TWSR&0xF8;
    if(!twi.phase) return;
    /* Recuperación I2C cooperativa: un paso por vencimiento; SCL bajo por salida, alto al liberar el pin con pull-up externo. */
    if(twi.phase>=5) {
        if(!due(n,twi.at)) return;
        twi.at=n+1;
        if(twi.phase==5) {
            DDRC|=_BV(PC5); twi.phase=6;
        } else if(twi.phase==6) {
            DDRC&=(uint8_t)~_BV(PC5);
            if(++twi.pulses<9) twi.phase=5;
            else { DDRC|=_BV(PC4); twi.phase=7; }
        } else {
            DDRC&=(uint8_t)~(_BV(PC4)|_BV(PC5));
            TWCR=_BV(TWEN); twi.phase=0; twi.ok=0;
        }
        return;
    }
    /* El envío solo se considera exitoso cuando el hardware termina STOP; un timeout inicia recuperación. */
    if(twi.phase==4) {
        if(!(TWCR&_BV(TWSTO))) { twi.phase=0; twi.ok=1; }
        else if(due(n,twi.at)) twi_abort(n);
        return;
    }
    if(!(TWCR&_BV(TWINT))) { if(due(n,twi.at)) twi_abort(n); return; }
    /* Secuencia normal: START aceptado, dirección PCF8574 en escritura, dato reconocido y STOP. */
    if(twi.phase==1 && (s==0x08 || s==0x10)) {
        TWDR=LCD_ADDR_7BIT<<1; TWCR=_BV(TWINT)|_BV(TWEN); twi.phase=2;
    } else if(twi.phase==2 && s==0x18) {
        TWDR=twi.value; TWCR=_BV(TWINT)|_BV(TWEN); twi.phase=3;
    } else if(twi.phase==3 && s==0x28) {
        TWCR=_BV(TWINT)|_BV(TWEN)|_BV(TWSTO); twi.phase=4;
    } else { twi_abort(n); return; }
    twi.at=n+4;
}

static char display[2][17];
static struct {
    uint8_t init,active,idx,len,packet[6],position,data_next;
    uint32_t at;
} lcd;
/* lcd_packet — Construye los bytes de PCF8574 necesarios para enviar un comando o carácter a la LCD.
 * Entradas: b, byte LCD; rs, selección de registro (0 comando, 1 dato); nibble, distinto de cero para enviar solo el nibble alto. Sin retorno. Incluye luz de fondo y pulsos de E; prepara tres bytes por nibble y activa el paquete para lcd_task.
 */
static void lcd_packet(uint8_t b,uint8_t rs,uint8_t nibble)
{
    uint8_t bl=LCD_BACKLIGHT_HIGH ? 8:0;
    uint8_t hi=(b&0xF0)|bl|rs,lo=(b<<4)|bl|rs;
    /* Cada nibble se presenta con E bajo, E alto y E bajo; RW permanece bajo en estos bytes. */
    lcd.packet[0]=hi; lcd.packet[1]=hi|4; lcd.packet[2]=hi;
    lcd.packet[3]=lo; lcd.packet[4]=lo|4; lcd.packet[5]=lo;
    lcd.len=nibble ? 3:6; lcd.idx=0; lcd.active=1;
}
/* lcd_task — Inicializa y refresca por pasos la LCD HD44780 de 16 columnas y dos filas.
 * Entrada: n, tiempo actual en ms. Sin retorno. Espera al TWI sin bloquear, transmite los paquetes, aplica los tiempos programados y recorre los 32 caracteres de display alternando dirección y dato. Si falla un envío, reinicia la secuencia LCD tras 1000 ms.
 */
static void lcd_task(uint32_t n)
{
    static const uint8_t init_cmd[]={0x30,0x30,0x30,0x20,0x28,0x08,0x01,0x06,0x0C};
    if(twi.phase) return;
    if(lcd.active) {
        /* Un error TWI invalida la inicialización y el refresco LCD; se espera antes de intentar la secuencia completa. */
        if(lcd.idx && !twi.ok) {
            lcd.active=0; lcd.init=0; lcd.position=0; lcd.data_next=0;
            lcd.at=n+1000; return;
        }
        if(lcd.idx<lcd.len) { twi_start(lcd.packet[lcd.idx++],n); return; }
        lcd.active=0; lcd.at=n+(lcd.init==1 ? 5:2); return;
    }
    if(!due(n,lcd.at)) return;
    /* Arranque HD44780: primeros cuatro envíos de un nibble; los siguientes comandos usan los dos nibbles. */
    if(lcd.init<sizeof(init_cmd)) {
        uint8_t i=lcd.init++; lcd_packet(init_cmd[i],0,i<4); return;
    }
    /* Refresco: posiciones 0..15 en primera fila y 16..31 desde DDRAM 0x40 en segunda fila; alterna dirección y carácter. */
    if(!lcd.data_next) {
        lcd_packet(0x80|(lcd.position<16 ? lcd.position:0x40+lcd.position-16),0,0);
        lcd.data_next=1;
    } else {
        lcd_packet(display[lcd.position/16][lcd.position%16],1,0);
        lcd.position=(lcd.position+1)%32; lcd.data_next=0;
    }
}
/* state_name — Obtiene el texto de presentación del estado actual.
 * Entrada: ninguna; consulta state, emergencia física, sensor_fault, thermal y piece. Retorno: puntero a una cadena constante. En estados de alarma distingue la condición activa de la indicación de rearme, manteniendo exactamente los textos originales.
 */
static const char *state_name(void)
{
    switch(state) {
    case S_RUN:return "MARCHA"; case S_PISTON:return "PISTON";
    case S_EMERGENCY:return (PIND&_BV(PD3)) ? "EMERG: DAR START":"EMERG BOTON";
    case S_THERMAL:return sensor_fault ? "FALLO SENSOR NTC":
                                 (thermal ? "EMERG TEMP":"TEMP OK START");
    case S_JAM:return piece ? "ATASCO PIEZA":"LDR OK DAR START";
    case S_LINK:return "FALLO SPI"; default:return "PARADO";
    }
}
/* line — Carga una fila del búfer de pantalla con ancho fijo de 16 caracteres.
 * Entradas: row, fila usada por el programa (0 o 1); s, cadena terminada en cero. Sin retorno. Copia hasta 16 caracteres, rellena con espacios y agrega el terminador cero en la posición 16. No comprueba el rango de row.
 */
static void line(uint8_t row,const char *s)
{
    uint8_t i=0;
    while(i<16 && s[i]) { display[row][i]=s[i]; ++i; }
    while(i<16) display[row][i++]=' ';
    display[row][16]=0;
}
/* report — Actualiza las filas de la LCD y genera telemetría UART cada 500 ms.
 * Entrada: n, tiempo actual en ms. Sin retorno. Informa diagnóstico SPI y luego estado, alarmas o páginas normales alternadas cada 2 s. La LCD limita el contador a módulo 100000 para su presentación; el informe UART incluye el contador completo, sensores, PWM y último comando confirmado. uart_text puede descartar un mensaje si no cabe en su cola.
 */
static void report(uint32_t n)
{
    static uint32_t next;
    char a[48],msg[220]; unsigned int percent=(pwm_target*100U+127)/255;
    if(!due(n,next)) return;
    /* Cadencia del informe: prepara diagnóstico SPI y filas LCD; luego encola telemetría con ambas filas y sensores. */
    next=n+500;
    snprintf(msg,sizeof(msg),"SPI C=%02X RX=%02X,%02X %s\r\n",debug_cmd,debug_rx0,debug_rx1,link_ok ? "OK":"FALLO");
    uart_text(msg);
    if(!ready_sensors) { line(0,"INICIALIZANDO"); line(1,"SENSORES / SPI"); }
    /* Las alarmas ocupan las filas con el motivo y la acción de recuperación según las condiciones presentes. */
    else if(state==S_EMERGENCY || state==S_THERMAL || state==S_JAM || state==S_LINK) {
        line(0,state_name());
        if(emergency_pressed()) line(1,"LIBERAR EMERG");
        else if(thermal) line(1,"REVISAR TEMP/NTC");
        else if(jam_latched && piece) line(1,"LIBERAR LDR");
        else if(!link_ok || need_sync) line(1,"REVISAR ENLACE");
        else line(1,"PULSAR START");
    /* Sin alarma, alterna cada 2 s entre ADC/contador y estado/temperatura/velocidad. */
    } else if((n/2000)&1) {
        snprintf(a,sizeof(a),"L%04u N%04u",analog[0],analog[1]); line(0,a);
        snprintf(a,sizeof(a),"C:%05lu %s",(unsigned long)(count%100000UL),last_command); line(1,a);
    } else {
        line(0,state_name());
        if(sensor_fault) snprintf(a,sizeof(a),"NTC ERROR V%3u%%",percent);
        else snprintf(a,sizeof(a),"T:%3dC V:%3u%%",temperature(),percent);
        line(1,a);
    }
    snprintf(msg,sizeof(msg),
        "LCD|%s|%s| EST=%s LDR=%u PIEZA=%u NTC=%u T=%dC POT=%u PWM=%u CNT=%lu CMD=%s SPI=%s\r\n",
        display[0],display[1],state_name(),analog[0],piece,analog[1],temperature(),
        analog[2],pwm_target,(unsigned long)count,last_command,link_ok ? "OK":"FALLO");
    uart_text(msg);
}
/* main — Configura el ATmega328P y ejecuta continuamente las tareas cooperativas del maestro.
 * Entrada: ninguna. Retorno declarado: int; el bucle infinito no retorna. Inicializa watchdog, pines, SPI, ADC, TWI a 100 kHz, UART 9600 8N1 y Timer2 de 1 ms; habilita interrupciones y watchdog de 1 s. Conserva el orden de llamadas para adquirir sensores, evaluar entradas, avanzar SPI, controlar proceso, atender TWI/LCD e informar.
 */
int main(void)
{
    /* Arranque: limpia la causa de reset y deshabilita el watchdog antes de configurar periféricos. */
    MCUSR=0; wdt_disable();
    /* PD2 START y PD3 emergencia son entradas con pull-up interno; los pulsadores se activan a GND. */
    DDRD&=(uint8_t)~(_BV(PD2)|_BV(PD3)); PORTD|=_BV(PD2)|_BV(PD3);
    /* SPI: SS, MOSI y SCK como salidas; MISO como entrada. SS se mantiene alto al restaurar el periférico. */
    DDRB|=_BV(PB2)|_BV(PB3)|_BV(PB5); DDRB&=(uint8_t)~_BV(PB4);
    PORTB|=_BV(PB2); spi_restore();
    /* ADC0..2 e I2C: entradas sin pull-up interno; ADC con referencia AVCC y prescaler 128, entradas digitales ADC0..2 deshabilitadas. */
    DDRC&=(uint8_t)~0x3F; PORTC&=(uint8_t)~0x3F;
    ADMUX=_BV(REFS0); ADCSRA=_BV(ADEN)|_BV(ADPS2)|_BV(ADPS1)|_BV(ADPS0);
    DIDR0=_BV(ADC0D)|_BV(ADC1D)|_BV(ADC2D);
    /* TWI: prescaler 1 y TWBR=72 para 100 kHz con F_CPU=16 MHz; SDA/SCL necesitan pull-ups externos. */
    TWSR=0; TWBR=72; TWCR=_BV(TWEN); /* I2C 100 kHz */
    /* UART: divisor 103, modo normal, 9600 baudios aproximados, 8 bits, sin paridad y un bit de parada; recepción habilitada sin procesar comandos. */
    UBRR0H=0; UBRR0L=103; UCSR0A=0;
    UCSR0B=_BV(TXEN0)|_BV(RXEN0); UCSR0C=_BV(UCSZ01)|_BV(UCSZ00);
    /* Timer2 en CTC, prescaler 64 y OCR2A=249: interrupción de comparación cada 1 ms a 16 MHz. */
    TCCR2A=_BV(WGM21); TCCR2B=_BV(CS22); OCR2A=249; TIMSK2=_BV(OCIE2A);
    line(0,"INICIALIZANDO"); line(1,"SENSORES / SPI"); lcd.at=50;
    /* Habilita las interrupciones y el watchdog; cada recorrido del bucle lo reinicia. */
    sei(); wdt_enable(WDTO_1S);
    for(;;) {
        uint32_t n=now_ms(); wdt_reset();
        /* Orden cooperativo: adquisición, seguridad/entradas, transporte SPI y control; cada tarea conserva su estado entre llamadas. */
        sensors(n); inputs(n); spi_task(n); control(n);
        /* Después se atienden I2C, refresco LCD e informes; las esperas se resuelven en llamadas posteriores. */
        twi_task(n); lcd_task(n); report(n);
    }
}
